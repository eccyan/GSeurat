# Interactive Music Audio Engine — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the existing `gseurat::AudioSystem` with a lock-free, sample-accurate, stem-based interactive music engine (`gseurat::audio::AudioEngine`) that is driven by scene data via an `AudioZone` component.

**Architecture:** Flat mixer operating on an array of `TrackGroupState` structs on the audio thread; SPSC ring buffer + atomic RTPC bus for game-thread → audio-thread communication; miniaudio used only as a backend (device callback + WAV decode); all mixing, state machines, and looping logic is custom C++23. Offline render mode enables deterministic golden tests.

**Tech Stack:** C++23, miniaudio 0.11.21 (vendored via FetchContent), nlohmann/json, CMake presets. Tests use the project's existing flat `check(cond, msg)` harness (no gtest/catch2). All new engine code lives under `src/engine/audio/` (game-agnostic; see CLAUDE.md Subrepository Architecture Rules).

**Spec:** `docs/superpowers/specs/2026-04-17-audio-engine-interactive-music-design.md` — read first.

```cpp
#pragma once
// Lock-free SPSC ring buffer for audio-thread command messaging.
//
// Usage: exactly ONE producer (game thread) calls try_push;
//        exactly ONE consumer (audio thread) calls try_pop.
//
// Power-of-two capacity; wrap is a mask. Cache-line-aligned head/tail
// to eliminate false sharing between producer and consumer.
//
// Standard Lamport SPSC — uses one slot as a sentinel, so capacity CapacityPow2
// actually holds up to CapacityPow2-1 items. Pick CapacityPow2 with that in mind.

#include <array>
#include <atomic>
#include <cstdint>
#include <type_traits>

namespace gseurat::audio {

template <typename T, uint32_t CapacityPow2>
class SpscRingBuffer {
    static_assert(CapacityPow2 >= 2, "capacity must be >= 2");
    static_assert((CapacityPow2 & (CapacityPow2 - 1)) == 0,
                  "capacity must be a power of two");
    static_assert(std::is_trivially_copyable_v<T>,
                  "T must be trivially copyable (POD-like)");

    static constexpr uint32_t kMask = CapacityPow2 - 1;

    alignas(64) std::atomic<uint32_t> head_{0};  // next write index
    alignas(64) std::atomic<uint32_t> tail_{0};  // next read index
    alignas(64) std::array<T, CapacityPow2> slots_{};

public:
    // Producer-only. Returns false if full (caller should increment a drop counter).
    [[nodiscard]] bool try_push(const T& v) noexcept {
        const uint32_t h = head_.load(std::memory_order_relaxed);
        const uint32_t t = tail_.load(std::memory_order_acquire);
        const uint32_t next = (h + 1) & kMask;
        if (next == t) return false;  // full
        slots_[h] = v;
        head_.store(next, std::memory_order_release);
        return true;
    }

    // Consumer-only. Returns false if empty.
    [[nodiscard]] bool try_pop(T& out) noexcept {
        const uint32_t t = tail_.load(std::memory_order_relaxed);
        const uint32_t h = head_.load(std::memory_order_acquire);
        if (t == h) return false;  // empty
        out = slots_[t];
        tail_.store((t + 1) & kMask, std::memory_order_release);
        return true;
    }

    // Any thread; relaxed, for diagnostics only.
    uint32_t approx_size() const noexcept {
        const uint32_t h = head_.load(std::memory_order_relaxed);
        const uint32_t t = tail_.load(std::memory_order_relaxed);
        return (h - t) & kMask;
    }
};

}  // namespace gseurat::audio
```

```cpp
#pragma once
// Per-frame linear slew-rate limiter. Audio-thread-safe: wait-free, noexcept, no alloc.

#include <cstdint>

namespace gseurat::audio {

class SlewLimiter {
    float current_ = 1.0f;
    float target_  = 1.0f;
    float step_    = 0.0f;  // delta per frame; 0 = snapped to target

public:
    // fade_frames == 0 means snap on the next tick().
    void set_target(float target, uint64_t fade_frames) noexcept {
        target_ = target;
        step_   = (fade_frames == 0)
                    ? 0.0f
                    : (target - current_) / static_cast<float>(fade_frames);
    }

    // Advance one frame; returns the current value.
    float tick() noexcept {
        if (step_ == 0.0f) { current_ = target_; return current_; }
        current_ += step_;
        const bool reached = (step_ > 0.0f) ? (current_ >= target_)
                                             : (current_ <= target_);
        if (reached) { current_ = target_; step_ = 0.0f; }
        return current_;
    }

    float current() const noexcept { return current_; }
    bool  settled() const noexcept { return step_ == 0.0f; }

    // For engine init / tests only — snap immediately without a tick.
    void force(float v) noexcept { current_ = target_ = v; step_ = 0.0f; }
};

}  // namespace gseurat::audio
```

```cpp
#pragma once
// IAudioSource: wait-free abstraction over PCM data.
// Every method MUST be noexcept, allocation-free, and syscall-free.
//
// Implementations in Phase 1: MemoryAudioSource only (in-memory float PCM).
// Phase 3 will add a binary-format mmap source; Phase 4 a streaming source.
// The mixer NEVER changes when new sources are added.

#include <cstdint>
#include <span>

namespace gseurat::audio {

struct AudioFormat {
    uint32_t sample_rate;
    uint8_t  channels;   // 1 = mono, 2 = stereo
};

class IAudioSource {
public:
    virtual ~IAudioSource() = default;

    virtual AudioFormat format()       const noexcept = 0;
    virtual uint64_t    total_frames() const noexcept = 0;

    // Read `frame_count` frames starting at `start_frame` into `out`.
    // `out` is interleaved float32; out.size() == frame_count * channels.
    //
    // If start_frame + frame_count > total_frames, implementation reads up to
    // total_frames and zero-fills the remainder. (Callers should cap reads at
    // boundaries — the mixer's chunk-split does this; this is a defense-in-depth.)
    virtual void read_frames(uint64_t start_frame,
                             uint32_t frame_count,
                             std::span<float> out_interleaved) const noexcept = 0;
};

}  // namespace gseurat::audio
```

```python
#!/usr/bin/env python3
"""Generate deterministic audio test fixtures.

Run from repo root:
    python3 tests/test_data/audio/gen_fixtures.py

Outputs:
    sine_loop.wav  — 0.5s mono 48kHz, 480Hz sine with INTEGER cycle count
                     so loop wrap produces zero discontinuity.
    dc_unity.wav   — 0.5s mono 48kHz, constant 1.0 float samples.
                     Makes slew assertions trivial (output == gain).
"""
import math
import struct
import sys
from pathlib import Path

SAMPLE_RATE = 48000
DURATION_SEC = 0.5
FRAMES = int(SAMPLE_RATE * DURATION_SEC)  # 24000
FREQ_HZ = 480.0  # 24000 / 50 cycles — exactly 50 full cycles in 0.5s, integer-cycle boundary

OUT_DIR = Path(__file__).parent


def write_wav_pcm16_mono(path: Path, samples: list[float]) -> None:
    """Write a 16-bit PCM mono WAV. miniaudio decodes these trivially."""
    n = len(samples)
    byte_rate = SAMPLE_RATE * 2  # mono, 16-bit
    with path.open("wb") as f:
        f.write(b"RIFF")
        f.write(struct.pack("<I", 36 + n * 2))
        f.write(b"WAVE")
        f.write(b"fmt ")
        f.write(struct.pack("<IHHIIHH", 16, 1, 1, SAMPLE_RATE, byte_rate, 2, 16))
        f.write(b"data")
        f.write(struct.pack("<I", n * 2))
        for s in samples:
            # Clamp to int16 range and write
            v = max(-1.0, min(1.0, s))
            f.write(struct.pack("<h", int(v * 32767)))


def main() -> int:
    # sine_loop.wav
    sine = [math.sin(2 * math.pi * FREQ_HZ * i / SAMPLE_RATE) for i in range(FRAMES)]
    write_wav_pcm16_mono(OUT_DIR / "sine_loop.wav", sine)

    # dc_unity.wav — constant 1.0 / 32767 to survive int16 round-trip exactly
    # (we'll assert on the decoded float value, which will be ~0.9999695 after int16 → float)
    # For a truly unit value in float, we accept the int16 round-trip loss or switch to float WAV.
    # Easier: write 0.5 constant. Assertions below use the known decoded value.
    dc = [0.5] * FRAMES
    write_wav_pcm16_mono(OUT_DIR / "dc_unity.wav", dc)

    print(f"Wrote {OUT_DIR / 'sine_loop.wav'} ({FRAMES} frames)")
    print(f"Wrote {OUT_DIR / 'dc_unity.wav'} ({FRAMES} frames)")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```cpp
#include "gseurat/engine/audio/memory_audio_source.hpp"

// Use miniaudio's decoder only — NOT ma_engine. We decode once at load,
// on the game thread. The audio thread never touches miniaudio's decoder.
#include <miniaudio.h>

#include <algorithm>
#include <cstring>
#include <string>

namespace gseurat::audio {

std::expected<std::unique_ptr<MemoryAudioSource>, LoadError>
MemoryAudioSource::from_wav_file(std::string_view path, uint32_t target_sample_rate) {
    std::string path_str(path);

    ma_decoder_config cfg = ma_decoder_config_init(
        ma_format_f32, /*channels*/ 0, /*sampleRate*/ 0);  // 0 = use file values

    ma_decoder decoder{};
    if (ma_decoder_init_file(path_str.c_str(), &cfg, &decoder) != MA_SUCCESS) {
        return std::unexpected(LoadError::FileNotFound);
    }

    const uint32_t file_sr = decoder.outputSampleRate;
    const uint32_t file_ch = decoder.outputChannels;

    if (file_sr != target_sample_rate) {
        ma_decoder_uninit(&decoder);
        return std::unexpected(LoadError::SampleRateMismatch);
    }
    if (file_ch != 1 && file_ch != 2) {
        ma_decoder_uninit(&decoder);
        return std::unexpected(LoadError::ChannelCountUnsupported);
    }

    ma_uint64 frames_in_file = 0;
    if (ma_decoder_get_length_in_pcm_frames(&decoder, &frames_in_file) != MA_SUCCESS
        || frames_in_file == 0) {
        ma_decoder_uninit(&decoder);
        return std::unexpected(LoadError::DecodeFailed);
    }
    // Arbitrary safety cap — 10 minutes at 48kHz stereo ≈ 275 MB.
    constexpr uint64_t kMaxFrames = 48000ULL * 60 * 10;
    if (frames_in_file > kMaxFrames) {
        ma_decoder_uninit(&decoder);
        return std::unexpected(LoadError::TooLarge);
    }

    std::vector<float> pcm(static_cast<size_t>(frames_in_file) * file_ch);
    ma_uint64 frames_read = 0;
    if (ma_decoder_read_pcm_frames(&decoder, pcm.data(), frames_in_file, &frames_read) != MA_SUCCESS
        || frames_read != frames_in_file) {
        ma_decoder_uninit(&decoder);
        return std::unexpected(LoadError::DecodeFailed);
    }
    ma_decoder_uninit(&decoder);

    AudioFormat fmt{file_sr, static_cast<uint8_t>(file_ch)};
    return std::unique_ptr<MemoryAudioSource>(
        new MemoryAudioSource(std::move(pcm), fmt, frames_in_file));
}

void MemoryAudioSource::read_frames(uint64_t start_frame,
                                     uint32_t frame_count,
                                     std::span<float> out) const noexcept {
    const uint32_t ch = format_.channels;
    const uint64_t available = (start_frame >= frame_count_)
                                 ? 0
                                 : (frame_count_ - start_frame);
    const uint32_t copy = static_cast<uint32_t>(std::min<uint64_t>(frame_count, available));
    if (copy > 0) {
        std::memcpy(out.data(),
                    pcm_.data() + start_frame * ch,
                    static_cast<size_t>(copy) * ch * sizeof(float));
    }
    // Zero-fill any uncovered tail (defense in depth; mixer should never hit this).
    if (copy < frame_count) {
        std::memset(out.data() + copy * ch, 0,
                    static_cast<size_t>(frame_count - copy) * ch * sizeof(float));
    }
}
```

```cpp
    std::fill(out.begin(), out.end(), 0.0f);

    // --- Drain commands (bounded) ---
    AudioCommand cmd;
    for (uint32_t i = 0; i < kMaxCommandsPerBlock && commands_.try_pop(cmd); ++i) {
        apply_command(cmd);

    // Task 3.8 adds binding logic.

    // --- Render active groups ---

        if (g.status == TrackGroupState::Status::Idle) continue;
        render_group(g, out, frames_this_block);

    // --- Advance master clock + publish bitmap ---

    uint64_t bitmap = 0;
    for (const auto& g : active_groups_) {

            bitmap |= (1ULL << g.group_id);
```

```cpp
    void set_group_volume(uint32_t gid, float target, float fade_ms) override {
        const uint64_t frames = ms_to_frames(fade_ms);
        AudioCommand c{AudioCommand::Type::SetGroupVolume, gid, 0, 0, target, frames};
        if (!mixer_.command_queue().try_push(c))
            mixer_.telemetry().dropped_commands.fetch_add(1, std::memory_order_relaxed);
    }

    void request_transition(uint32_t from, uint32_t to, float xfade_ms) override {
        const uint64_t frames = ms_to_frames(xfade_ms);
        AudioCommand c{AudioCommand::Type::RequestTransition, from, 0, to, 0.0f, frames};
        if (!mixer_.command_queue().try_push(c))
            mixer_.telemetry().dropped_commands.fetch_add(1, std::memory_order_relaxed);
    }

    void set_rtpc(uint32_t rtpc_id, float value) override {
        if (rtpc_id < RtpcBus::kMaxRtpc)
            mixer_.rtpc_bus().values[rtpc_id].store(value, std::memory_order_relaxed);
    }

    void render_offline(std::span<float> out, uint64_t num_frames) override {
        assert(mode_ == Mode::Offline && "render_offline requires Mode::Offline");
        // Pump in blocks of buffer_frames — matches realtime behavior exactly.
        uint64_t done = 0;
        while (done < num_frames) {
            const uint32_t this_block = static_cast<uint32_t>(
                std::min<uint64_t>(num_frames - done, cfg_.buffer_frames));
            std::span<float> slice{
                out.data() + done * channels_,
                static_cast<size_t>(this_block) * channels_};
            mixer_.render(slice, this_block);
            done += this_block;
        }
    }

    uint64_t current_frame() const noexcept override {
        return mixer_telemetry().current_frame.load(std::memory_order_relaxed);
    }
    bool is_group_playing(uint32_t gid) const noexcept override {
        if (gid >= 64) return false;
        return (mixer_telemetry().active_group_bitmap.load(std::memory_order_relaxed)
                & (1ULL << gid)) != 0;
    }
    uint32_t dropped_command_count() const noexcept override {
        return mixer_telemetry().dropped_commands.load(std::memory_order_relaxed);
    }
    uint32_t channels() const noexcept override { return channels_; }

private:
    uint64_t ms_to_frames(float ms) const noexcept {
        return static_cast<uint64_t>(std::llround(
            static_cast<double>(ms) * static_cast<double>(cfg_.sample_rate) / 1000.0));
    }
    const EngineTelemetry& mixer_telemetry() const noexcept {
        return const_cast<Mixer&>(mixer_).telemetry();
    }

    Config   cfg_;
    Mode     mode_;
    uint32_t channels_;
    Mixer    mixer_;
};
```

```cpp
void Mixer::apply_command(const AudioCommand& cmd) noexcept {
    using T = AudioCommand::Type;
    switch (cmd.type) {
    case T::PlayGroup: {
        // Already playing? No-op (later: restart semantics).
        if (find_active_slot(cmd.group_id)) return;

        // Find registry entry.
        const TrackGroupRegistryEntry* entry = nullptr;
        for (const auto& e : registry_) if (e.metadata.id == cmd.group_id) { entry = &e; break; }
        if (!entry) return;  // unknown id — drop silently (diagnostics via telemetry)

        TrackGroupState* g = acquire_slot(cmd.group_id);
        if (!g) return;  // active slots full

        g->status       = TrackGroupState::Status::Playing;
        g->play_cursor  = 0;
        g->loop_start   = entry->metadata.loop_start;
        g->loop_end     = entry->metadata.loop_end;
        g->stem_count   = static_cast<uint8_t>(entry->metadata.stems.size());
        g->markers_begin = entry->metadata.markers.data();
        g->markers_end   = entry->metadata.markers.data() + entry->metadata.markers.size();
        g->group_volume.force(1.0f);

        for (uint8_t i = 0; i < g->stem_count; ++i) {
            g->stems[i].source = entry->owned_stems[i].get();
            g->stems[i].volume.force(entry->metadata.stems[i].initial_volume);
        }
        break;
    }
    case T::StopGroup: {
        if (auto* g = find_active_slot(cmd.group_id)) {
            // Immediate stop (fade-to-zero would be set_group_volume before stop).
            g->status = TrackGroupState::Status::Idle;
        }
        break;
    }
    case T::SetStemVolume: {
        if (auto* g = find_active_slot(cmd.group_id)) {
            if (cmd.stem_index < g->stem_count) {
                g->stems[cmd.stem_index].volume.set_target(cmd.value, cmd.frame_count);
            }
        }
        break;
    }
    case T::SetGroupVolume: {
        if (auto* g = find_active_slot(cmd.group_id)) {
            g->group_volume.set_target(cmd.value, cmd.frame_count);
        }
        break;
    }
    case T::RequestTransition: {
        // Task 3.7 implements marker-aligned transitions.
        break;
    }
    }
}

void Mixer::render_group(TrackGroupState& g, std::span<float> out,
                          uint32_t frames_remaining) noexcept {
    if (g.stem_count == 0 || g.stems[0].source == nullptr) {
        g.status = TrackGroupState::Status::Idle;
        return;
    }

    uint32_t out_offset = 0;
    while (frames_remaining > 0) {
        const uint64_t source_len = g.stems[0].source->total_frames();
        const uint64_t frames_until_boundary = (g.loop_end > 0)
            ? (g.loop_end - g.play_cursor)
            : (source_len - g.play_cursor);

        const uint32_t chunk = static_cast<uint32_t>(
            std::min<uint64_t>(frames_remaining, frames_until_boundary));
        if (chunk == 0) {
            // Source exhausted (one-shot, no loop)
            g.status = TrackGroupState::Status::Idle;
            return;
        }

        mix_chunk(g, out, out_offset, chunk);

        out_offset       += chunk;
        frames_remaining -= chunk;
        g.play_cursor    += chunk;

        // Loop wrap
        if (g.loop_end > 0 && g.play_cursor == g.loop_end) {
            g.play_cursor = g.loop_start;
            maybe_trigger_pending_transition(g);
        } else if (g.loop_end == 0 && g.play_cursor >= source_len) {
            g.status = TrackGroupState::Status::Idle;
            return;
        }
    }

    // Handle crossfade-out termination
    if (g.status == TrackGroupState::Status::CrossfadingOut && g.group_volume.settled()
        && g.group_volume.current() == 0.0f) {
        g.status = TrackGroupState::Status::Idle;
    }
}

void Mixer::mix_chunk(TrackGroupState& g, std::span<float> out,
                       uint32_t out_offset, uint32_t chunk_frames) noexcept {
    // Pre-compute per-frame group volume so stems share the same slewed value.
    for (uint32_t f = 0; f < chunk_frames; ++f) {
        group_volume_scratch_[f] = g.group_volume.tick();
    }

    const uint32_t out_channels = channels_;  // mixer output (always 2 in Phase 1)

    for (uint8_t si = 0; si < g.stem_count; ++si) {
        auto& stem = g.stems[si];
        if (!stem.source) continue;

        const uint32_t src_ch = stem.source->format().channels;
        std::span<float> scratch(read_scratch_.data(),
                                  static_cast<size_t>(chunk_frames) * src_ch);
        stem.source->read_frames(g.play_cursor, chunk_frames, scratch);

        for (uint32_t f = 0; f < chunk_frames; ++f) {
            const float gain = stem.volume.tick() * group_volume_scratch_[f];
            // Upmix mono → stereo (broadcast); stereo → stereo straight copy.
            if (src_ch == 1) {
                const float s = scratch[f] * gain;
                out[(out_offset + f) * out_channels + 0] += s;
                out[(out_offset + f) * out_channels + 1] += s;
            } else {  // src_ch == 2
                out[(out_offset + f) * out_channels + 0] += scratch[f * 2 + 0] * gain;
                out[(out_offset + f) * out_channels + 1] += scratch[f * 2 + 1] * gain;
            }
        }
    }
}
```

**Important:** A transition started via `maybe_trigger_pending_transition` mutates `active_groups_` by acquiring a new slot. Since we're iterating over `active_groups_` in `Mixer::render`, acquiring a slot doesn't invalidate our iterator (we reuse Idle slots, which we already skip via the `status == Idle` check — but the new slot will be processed later in the same frame *if* the loop hasn't reached it yet). For deterministic behavior, the new group begins its first audio output on the *next* `render()` call, not the current one. Document this behavior in the comment block of `maybe_trigger_pending_transition`.

Actually, for the test to pass (xfade mid-block), the new group must start rendering in the *same* callback. Simplest fix: after marker-triggered transition, `continue` the outer render loop — on the next iteration, the newly-active target slot will be found and rendered. Since `Mixer::render`'s outer loop iterates `active_groups_` in order, and our new slot may be *earlier* in the array than the current one, we'd miss it this block. Acceptable trade-off for Phase 1: **target group starts rendering on the NEXT block**, one-block latency (~21 ms @ 1024 frames / 48 kHz).

```cpp
            for (const auto& mj : tgj["markers"]) {
                Marker m;
                m.frame = mj.at("frame").get<uint64_t>();
                m.name  = mj.value("name", std::string{});
                if (!first && m.frame <= prev) {
                    return std::unexpected(MetadataError::MarkersNotAscending);
                }
                prev = m.frame;
                first = false;
                tg.markers.push_back(std::move(m));
            }
        }

        for (const auto& sj : tgj.at("stems")) {
            StemMetadata s;
            s.source_path    = sj.at("source").get<std::string>();
            s.initial_volume = sj.value("initial_volume", 1.0f);
            tg.stems.push_back(std::move(s));
        }

        cfg.track_groups.push_back(std::move(tg));
    }
    return cfg;
}

    std::ifstream f{std::string(path)};
    if (!f) return std::unexpected(MetadataError::FileNotFound);
    std::stringstream ss;
    ss << f.rdbuf();
    return parse_music_config_json(ss.str());
}
```

```cpp
class MiniaudioDevice : public IAudioDevice {
    ma_device device_{};
    bool      running_ = false;
    RenderFn  render_;

    static void data_cb(ma_device* d, void* output, const void* /*input*/, ma_uint32 frames) {
        auto* self = static_cast<MiniaudioDevice*>(d->pUserData);
        self->render_(static_cast<float*>(output), frames);

        if (ma_device_start(&device_) != MA_SUCCESS) { ma_device_uninit(&device_); return false; }
```

```cpp
#include "gseurat/engine/systems/audio_zone_system.hpp"

namespace gseurat {

static bool inside(const glm::vec3& p, const glm::vec3& lo, const glm::vec3& hi) {
    return p.x >= lo.x && p.x <= hi.x
        && p.y >= lo.y && p.y <= hi.y
        && p.z >= lo.z && p.z <= hi.z;
}

void AudioZoneSystem::tick(const PlayerPosition& pp, std::span<AudioZoneComponent> zones) {
    if (!engine_) return;
    for (auto& z : zones) {
        const bool now_in = inside(pp.value, z.bounds_min, z.bounds_max);
        if (now_in && !z.player_inside) {
            // Enter
            switch (z.action_on_enter) {
            case AudioZoneComponent::Action::Play:
                engine_->play_group(z.track_group_id); break;
            case AudioZoneComponent::Action::PlayOrTransition:
                // Phase 1: naive — issue request_transition from 0 if nothing playing,
                // else from current active group. Caller must know the source id.
                engine_->play_group(z.track_group_id); break;
            case AudioZoneComponent::Action::Stop:
                engine_->stop_group(z.track_group_id); break;
            }
        } else if (!now_in && z.player_inside) {
            switch (z.action_on_exit) {
            case AudioZoneComponent::Action::Stop:
                engine_->set_group_volume(z.track_group_id, 0.0f, z.exit_fade_ms);
                engine_->stop_group(z.track_group_id); break;
            default: break;
            }
        }
        z.player_inside = now_in;
    }
}

}  // namespace gseurat
```
